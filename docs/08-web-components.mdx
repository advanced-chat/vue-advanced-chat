# Web Components

`@advanced-chat/components` exposes the complete chat surface as a
standards-based `<advanced-chat-components>` custom element. It bundles its own Vue
runtime, works in light DOM, and does not require the host application to use
Vue.

## Install

```bash
npm install @advanced-chat/components@next
```

Load both the auto-registering module and its stylesheet:

```ts
import '@advanced-chat/components/web-component'
import '@advanced-chat/components/web-component/styles'
```

Assign objects and arrays as DOM properties. Use attributes for simple strings,
numbers, and booleans only.

```html
<advanced-chat-components id="chat" height="600px" theme="auto"></advanced-chat-components>

<script type="module">
  import '@advanced-chat/components/web-component'
  import '@advanced-chat/components/web-component/styles'

  const chat = document.querySelector('#chat')
  const currentUser = {
    id: 'me',
    name: 'Alice',
    status: { state: 'online' },
  }

  chat.currentUser = currentUser
  chat.chats = [{ id: 'general', name: 'General', users: [currentUser] }]
  chat.chat = chat.chats[0]
  chat.messages = []
  chat.chatsLoaded = true
  chat.messagesLoaded = true

  chat.addEventListener('send-message', (event) => {
    console.log(event.detail)
  })
</script>
```

All public events are `CustomEvent` instances. Their payload is directly on
`event.detail`; events without a payload use `null`. Events have `bubbles: true`
and `composed: true`, so event delegation works from an application shell and
across host shadow roots.

## Explicit registration

The default entrypoint registers `<advanced-chat-components>` when imported in a
browser and safely skips registration when `customElements` is unavailable. Use
the side-effect-free core entrypoint when an application controls startup or uses
an alternate tag.

```ts
import {
  createAdvancedChatElement,
  registerAdvancedChat,
} from '@advanced-chat/components/web-component/core'
import '@advanced-chat/components/web-component/styles'

registerAdvancedChat({
  tagName: 'acme-chat',
  strings: {
    'chats.search.placeholder': 'Find a conversation',
  },
})

// Or manage the registry directly:
customElements.define('internal-chat', createAdvancedChatElement())
```

Call registration in a browser lifecycle hook when the application also runs
on a server. Calling `registerAdvancedChat` where `customElements` is not
available throws a descriptive error instead of failing implicitly.

## React

An imperative ref keeps the integration compatible across React versions and
retains typed DOM events.

```tsx
import { useEffect, useRef } from 'react'
import type { DetailedHTMLProps, HTMLAttributes } from 'react'
import '@advanced-chat/components/web-component'
import '@advanced-chat/components/web-component/styles'
import type { ChatModel, MessageModel, UserReference } from '@advanced-chat/components'
import type {
  AdvancedChatEventMap,
  AdvancedChatHTMLElement,
} from '@advanced-chat/components/web-component'

declare module 'react' {
  namespace JSX {
    interface IntrinsicElements {
      'advanced-chat-components': DetailedHTMLProps<
        HTMLAttributes<AdvancedChatHTMLElement>,
        AdvancedChatHTMLElement
      > & { height?: string }
    }
  }
}

interface ChatProps {
  currentUser: UserReference
  chats: ChatModel[]
  activeChat: ChatModel | null
  messages: MessageModel[]
  onSendMessage: (payload: AdvancedChatEventMap['send-message']['detail']) => void
}

export function Chat({ currentUser, chats, activeChat, messages, onSendMessage }: ChatProps) {
  const ref = useRef<AdvancedChatHTMLElement>(null)

  useEffect(() => {
    const element = ref.current
    if (!element) return

    element.currentUser = currentUser
    element.chats = chats
    element.chat = activeChat
    element.messages = messages
    element.chatsLoaded = true
    element.messagesLoaded = true

    const handleSend = (event: AdvancedChatEventMap['send-message']) => {
      onSendMessage(event.detail)
    }
    element.addEventListener('send-message', handleSend)
    return () => element.removeEventListener('send-message', handleSend)
  }, [currentUser, chats, activeChat, messages, onSendMessage])

  return <advanced-chat-components ref={ref} height="600px" />
}
```

React's TypeScript declarations do not automatically consume a Custom Elements
Manifest. The local augmentation above types the tag, standard attributes, and
`ref`; complex values still cross the boundary as DOM properties in the effect.

## Angular

Import the registration module once and permit custom elements in the owning
component or module.

```ts
import { Component, CUSTOM_ELEMENTS_SCHEMA } from '@angular/core'
import '@advanced-chat/components/web-component'
import '@advanced-chat/components/web-component/styles'
import type { ChatModel, MessageModel, UserReference } from '@advanced-chat/components'
import type { AdvancedChatEventMap } from '@advanced-chat/components/web-component'

@Component({
  selector: 'app-chat',
  standalone: true,
  schemas: [CUSTOM_ELEMENTS_SCHEMA],
  template: `
    <advanced-chat-components
      height="600px"
      [currentUser]="currentUser"
      [chats]="chats"
      [chat]="activeChat"
      [messages]="messages"
      [chatsLoaded]="true"
      [messagesLoaded]="true"
      (send-message)="send($event)"
    />
  `,
})
export class ChatComponent {
  readonly currentUser: UserReference = { id: 'me' }
  readonly chats: ChatModel[] = [
    {
      id: 'general',
      name: 'General',
      users: [
        { id: 'me', name: 'Alice', status: { state: 'online' } },
        { id: 'sam', name: 'Sam', status: { state: 'offline' } },
      ],
    },
  ]
  readonly activeChat = this.chats[0] ?? null
  readonly messages: MessageModel[] = []

  send(event: Event) {
    const payload = (event as AdvancedChatEventMap['send-message']).detail
    console.log(payload)
  }
}
```

## Svelte

Use `bind:this` when you want an explicit property and event boundary.

```svelte
<script lang="ts">
  import { onMount } from 'svelte'
  import '@advanced-chat/components/web-component'
  import '@advanced-chat/components/web-component/styles'
  import type { ChatModel, MessageModel, User } from '@advanced-chat/components'
  import type {
    AdvancedChatEventMap,
    AdvancedChatHTMLElement,
  } from '@advanced-chat/components/web-component'

  const currentUser: User = {
    id: 'me',
    name: 'Alice',
    status: { state: 'online' },
  }
  const chats: ChatModel[] = [{ id: 'general', name: 'General', users: [currentUser] }]
  const messages: MessageModel[] = []
  let chatElement: AdvancedChatHTMLElement

  onMount(() => {
    chatElement.currentUser = currentUser
    chatElement.chats = chats
    chatElement.chat = chats[0] ?? null
    chatElement.messages = messages
    chatElement.chatsLoaded = true
    chatElement.messagesLoaded = true

    const send = (event: AdvancedChatEventMap['send-message']) => console.log(event.detail)
    chatElement.addEventListener('send-message', send)
    return () => chatElement.removeEventListener('send-message', send)
  })
</script>

<advanced-chat-components bind:this={chatElement} height="600px" />
```

## Tooling and styles

The npm package publishes `custom-elements.json` using the Custom Elements
Manifest schema. Tooling that explicitly supports that schema can use it for
property, attribute, and event completion.

The element intentionally renders in light DOM. This keeps application-level
theming and CSS custom-property overrides straightforward, but host selectors
can affect the internal UI. Load exactly one matching web-component stylesheet
and scope broad application styles when necessary.

The distributed module targets modern browsers with native custom elements,
ES modules, dynamic imports, `IntersectionObserver`, and `ResizeObserver`. Add
platform polyfills before registration when supporting browsers without APIs
your integration relies on; without `ResizeObserver`, the component retains its
initial desktop/mobile layout calculation after mount.
